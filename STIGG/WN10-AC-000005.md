<# 

.SYNOPSIS
    Configures Windows 10 Account Lockout Policy.

.DESCRIPTION
    This PowerShell script sets the following Account Lockout Policy parameters:
    - Lockout Duration: 16 minutes
    - Lockout Threshold: 3 invalid attempts
    - Reset Lockout Counter: 15 minutes

.NOTES
    Author          : Giuseppe Scalzo
    LinkedIn        : https://www.linkedin.com/in/giuseppe-scalzo-/
    GitHub          : https://github.com/GScalzo21
    Date Created    : 2025-05-10
    Last Modified   : 2025-05-10
    Version         : 1.0
    CVEs            : N/A
    Plugin IDs      : N/A
    STIG-ID         : WN10-AC-000005

.TESTED ON
    Date(s) Tested  : 2025-05-10
    Tested By       : Giuseppe Scalzo
    Systems Tested  : Windows 10 (Local Group Policy)
    PowerShell Ver. : 5.1

.USAGE
    Run this script in PowerShell as Administrator.
    Example:
        PS C:\> .\Remediate_AccountLockoutPolicy.ps1
#>

# Export current security policy
secedit /export /cfg "$env:TEMP\secpol.cfg"

# Modify the necessary settings
$configPath = "$env:TEMP\secpol.cfg"

(Get-Content $configPath) |
    ForEach-Object {
        $_ -replace 'LockoutDuration\s*=\s*\d+', 'LockoutDuration = 16' `
           -replace 'LockoutBadCount\s*=\s*\d+', 'LockoutBadCount = 3' `
           -replace 'ResetLockoutCount\s*=\s*\d+', 'ResetLockoutCount = 15'
    } | Set-Content $configPath

# Apply the updated security policy
secedit /configure /db secedit.sdb /cfg $configPath /areas SECURITYPOLICY

# Clean up temporary file
Remove-Item $configPath -Force

# Output results
Write-Host "Account Lockout Policy successfully configured:" -ForegroundColor Green
Write-Host "- Lockout Duration: 16 minutes"
Write-Host "- Lockout Threshold: 3 attempts"
Write-Host "- Reset Counter: 15 minutes"
